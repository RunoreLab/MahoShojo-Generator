- [ ] **Step 3: 实现请求校验与 prompt 分层**

```ts
export function validateCreatorRequest(input) {
  if (input.buildRules.length > 0 && !input.primaryRuleId) throw new Error('PRIMARY_RULE_REQUIRED');
  if (input.buildRules.length > 0 && !input.buildRules.some((rule) => rule.ruleId === input.primaryRuleId)) {
    throw new Error('PRIMARY_RULE_NOT_SELECTED');
  }
  if (input.buildRules.length === 0 && input.questionnaires.length === 0 && !input.freeformBrief.trim()) {
    throw new Error('FREEFORM_BRIEF_REQUIRED');
  }
  for (const rule of input.buildRules) {
    const preset = getBuildRulePresetOrThrow(rule.ruleId);
    if (!preset.supportedTemplates.includes(input.template)) throw new Error('RULE_TEMPLATE_UNSUPPORTED');
    if (!preset.allowStandalone && input.questionnaires.length === 0) throw new Error('QUESTIONNAIRE_REQUIRED_FOR_RULE');
  }
  if (input.primaryRuleId) {
    const primaryPreset = getBuildRulePresetOrThrow(input.primaryRuleId);
    if (!primaryPreset.mainRuleEligible) throw new Error('PRIMARY_RULE_INELIGIBLE');
  }
}

export function buildCreatorPromptInput(input) {

    buildRuleProjection: projectBuildRulesForPrompt(input.buildStateLike),
  };
}
```
