# 创作页工作台布局 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 `/creator` 从窄屏单列页面重组为桌面双栏、移动单列的创作工作台，在不改动 creator 业务链路的前提下，把配置、导航与高级生成迁入左侧控制台，让右侧主区只保留开始、答题与结果三类主任务。

**Architecture:** 保留 [`pages/creator.tsx`](/home/notuhao/code/MahoShojo-Generator/pages/creator.tsx) 作为 state、effect 与提交逻辑的唯一持有者，新建 `components/creator/*` 布局壳与侧栏组件承接 UI 分层。布局层采用 `CreatorWorkbenchLayout + CreatorSidebar + CreatorMainStage`，通过 props 传入现有节点与回调，不新增独立业务状态源；样式层通过 `creator-workbench-shell` 作用域解除 `.container/.card` 的窄宽度限制，并只在 `lg` 及以上启用 sticky 左栏。

**Tech Stack:** Next.js Pages Router、React 19、TypeScript strict、Tailwind 4、Bun test、现有 `CollapsibleSection`、现有 creator/问卷/结果组件链路。

```tsx
// components/creator/CreatorWorkbenchLayout.tsx
type CreatorWorkbenchLayoutMode = 'desktop' | 'mobile';

export function CreatorWorkbenchLayout(props: {
  layoutMode: CreatorWorkbenchLayoutMode;
  sidebar: React.ReactNode;
  main: React.ReactNode;
}) {
  const mode = props.layoutMode;

  return (
    <div
      className="creator-workbench-shell mx-auto w-full px-4 pb-8 pt-6 sm:px-6 lg:max-w-[1200px] lg:px-6 xl:max-w-[1360px] xl:px-8 2xl:max-w-[1440px]"
      data-layout-mode={mode}
    >
      {mode === 'mobile' ? (
        <div className="creator-workbench-mobile space-y-4">
          <div className="creator-workbench-mobile-panels">{props.sidebar}</div>
          <div className="creator-workbench-main">{props.main}</div>
        </div>
      ) : (
        <div className="creator-workbench-grid grid gap-5 lg:grid-cols-[minmax(320px,340px)_minmax(0,1fr)] xl:grid-cols-[minmax(360px,400px)_minmax(0,1fr)]">
          <aside className="creator-workbench-sidebar min-w-0">{props.sidebar}</aside>
          <main className="creator-workbench-main min-w-0">{props.main}</main>
        </div>
      )}
    </div>
  );
}

// components/creator/CreatorMainStage.tsx
export function CreatorMainStage(props: {
  stage: 'status' | 'intro' | 'questionnaire' | 'result';
  title?: string;
  content: React.ReactNode;
}) {
  return (
    <section className="rounded-[28px] border p-5 sm:p-6 xl:p-8">
      {props.title ? <div className="mb-4 text-xl font-semibold text-slate-900">{props.title}</div> : null}
      <div data-creator-stage={props.stage}>{props.content}</div>
    </section>
  );
}
```

```tsx
import { CollapsibleSection } from '@/components/shared/CollapsibleSection';

const MOBILE_OPEN_BY_STAGE = {
  intro: { overview: true, configuration: true, questionnaire: false, advanced: false },
  questionnaire: { overview: true, configuration: false, questionnaire: true, advanced: false },
  result: { overview: true, configuration: false, questionnaire: false, advanced: false },

export function CreatorSidebar(props: {

  configuration: React.ReactNode;
  questionnaire: React.ReactNode;
  advanced: React.ReactNode;

  return (

      <div data-group="overview" data-default-open={String(isMobile ? defaults.overview : true)}>
        <CollapsibleSection title="创作概况" defaultOpen={isMobile ? defaults.overview : true} keepMounted>

        </CollapsibleSection>
```
